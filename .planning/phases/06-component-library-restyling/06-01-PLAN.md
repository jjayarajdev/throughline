**Step 1: Add shimmer keyframe to index.css** (per research recommendation to inline -- 200 bytes, used on every page):

Add before the `@layer base` block in `apps/web/src/index.css`:
```css
@keyframes shimmer {
  0% {
    background-position: -100% 0;
  }
  100% {
    background-position: 100% 0;
  }
}
```

Also add to the existing `@media (prefers-reduced-motion: reduce)` block (if it exists) or create one:
```css
@media (prefers-reduced-motion: reduce) {
  .skeleton-shimmer,
  [class*="animate-"] {
    animation: none !important;
    transition: none !important;
  }
}
```

**Step 2: Rewrite skeleton.tsx:**

```tsx
import * as React from 'react';
import { cn } from '@/lib/utils';

interface SkeletonProps extends React.HTMLAttributes<HTMLDivElement> {
  variant?: 'default' | 'text' | 'heading' | 'avatar' | 'button';
}

export function Skeleton({ className, variant = 'default', ...props }: SkeletonProps) {
  return (
    <div
      className={cn(
        'rounded-md bg-muted relative overflow-hidden',
        "before:absolute before:inset-0 before:-translate-x-full before:bg-gradient-to-r before:from-transparent before:via-muted-foreground/10 before:to-transparent before:animate-[shimmer_1.5s_ease-in-out_infinite]",
        variant === 'text' && 'h-4 w-full',
        variant === 'heading' && 'h-6 w-3/5',
        variant === 'avatar' && 'h-10 w-10 rounded-full',
        variant === 'button' && 'h-9 w-24 rounded-md',
        className,
      )}
      {...props}
    />
  );
}
```

Key changes from current:
- Removed `animate-pulse` -- replaced with shimmer via CSS `before` pseudo-element
- Added `variant` prop for common content shapes
- Uses `muted-foreground/10` for brand-cohesive shimmer gradient
- `overflow-hidden` contains the sliding gradient
- `prefers-reduced-motion` disables shimmer via global CSS rule

The Skeleton still exports as `Skeleton` with `className` and all HTMLDiv props -- no API breakage. The 14 files currently importing Skeleton will continue to work.
